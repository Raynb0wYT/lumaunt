cmake_minimum_required(VERSION 3.24)
project(LumauntDiscordSmoke LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Build the Discord connection harness on Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DISCORD_SDK_ROOT "" CACHE PATH "Windows Discord Social SDK root (include, lib/release, bin/release)")
set(DISCORD_ARCH_SUBDIR "")
# An explicit build target takes precedence over the ARM64 host processor.
if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64" OR
   (NOT CMAKE_GENERATOR_PLATFORM AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|arm64|aarch64)$"))
  set(DISCORD_ARCH_SUBDIR "/arm64")
endif()
set(DISCORD_LIB_DIR "${DISCORD_SDK_ROOT}/lib/release${DISCORD_ARCH_SUBDIR}")
set(DISCORD_BIN_DIR "${DISCORD_SDK_ROOT}/bin/release${DISCORD_ARCH_SUBDIR}")
foreach(required include/discordpp.h include/cdiscord.h "lib/release${DISCORD_ARCH_SUBDIR}/discord_partner_sdk.lib" "bin/release${DISCORD_ARCH_SUBDIR}/discord_partner_sdk.dll")
  if(NOT EXISTS "${DISCORD_SDK_ROOT}/${required}")
    message(FATAL_ERROR "Missing Windows SDK file: ${DISCORD_SDK_ROOT}/${required}")
  endif()
endforeach()
add_library(lumaunt_discord SHARED DiscordBridge.cpp exports.def)
target_include_directories(lumaunt_discord PRIVATE "${DISCORD_SDK_ROOT}/include")
target_link_libraries(lumaunt_discord PRIVATE "${DISCORD_LIB_DIR}/discord_partner_sdk.lib")
add_executable(lumaunt-discord-smoke smoke.cpp)
target_link_libraries(lumaunt-discord-smoke PRIVATE lumaunt_discord)
add_executable(lumaunt-artwork-check artwork-check.cpp)
target_include_directories(lumaunt-artwork-check PRIVATE "${DISCORD_SDK_ROOT}/include")
target_link_libraries(lumaunt-artwork-check PRIVATE "${DISCORD_LIB_DIR}/discord_partner_sdk.lib")
foreach(target lumaunt_discord lumaunt-discord-smoke lumaunt-artwork-check)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${DISCORD_BIN_DIR}/discord_partner_sdk.dll" "$<TARGET_FILE_DIR:${target}>/discord_partner_sdk.dll")
  if(EXISTS "${DISCORD_BIN_DIR}/discord_krisp.dll")
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${DISCORD_BIN_DIR}/discord_krisp.dll" "$<TARGET_FILE_DIR:${target}>/discord_krisp.dll")
  endif()
endforeach()

if(EXISTS "${DISCORD_SDK_ROOT}/License-Notices.txt")
  add_custom_command(TARGET lumaunt_discord POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${DISCORD_SDK_ROOT}/License-Notices.txt" "$<TARGET_FILE_DIR:lumaunt_discord>/License-Notices.txt")
endif()
